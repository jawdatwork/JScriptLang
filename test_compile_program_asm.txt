section .data
	hello:     db 'Compile program'    ; 'Hello, World!' plus a linefeed character
	helloLen:  equ $-hello             ; Length of the 'Hello world!' string
	myProgram: db 'int x = 1;'
	myProgramLen: equ $-myProgram
	newlineString: db '   ;   '
	newlineStringLen: equ $-newlineString
	bytecodeOutput: db '101, 102, 103'
	bytecodeOutputLen: equ $-bytecodeOutput

section .text
	global _start

_start:
	mov eax,4            ; The system call for write (sys_write)
	mov ebx,1            ; File descriptor 1 - standard output
	mov ecx,hello        ; Put the offset of hello in ecx
	mov edx,helloLen     ; helloLen is a constant, so we don't need to say
	                     ;  mov edx,[helloLen] to get it's actual val
	int 80h

    mov eax,4
	mov ebx,1
	mov ecx,newlineString
	mov edx,newlineStringLen
	int 80h

    mov eax,4
	mov ebx,1
	mov ecx,myProgram
	mov edx,myProgramLen
	int 80h              ; Call the kernel

	mov eax,4
	mov ebx,1
	mov ecx,newlineString
	mov edx,newlineStringLen
	int 80h

	mov eax,4
	mov ebx,1
	mov ecx,bytecodeOutput
	mov edx,bytecodeOutputLen
	int 80h

	mov eax,1            ; The system call for exit (sys_exit)
	mov ebx,0            ; Exit with return "code" of 0 (no error)
	int 80h;
